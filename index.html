# --- GEMINI FUNC ---
def get_gemini_response(user_input):
    headers = {"Content-Type": "application/json"}
    data = {
        "contents": [{"parts": [{"text": f"{BOT_PERSONALITY}\n\n:User  {user_input}\n\nAI:"}]}]
    }
    try:
        res = requests.post(GEMINI_API_URL, headers=headers, json=data)
        res.raise_for_status()
        result = res.json()
        candidates = result.get("candidates", [{}])
        if candidates:
            content = candidates[0].get("content", {})
            parts = content.get("parts", [{}])
            return parts[0].get("text", "No response found.")
        return "No response from AI."
    except Exception as e:
        return f"API Error: {str(e)}"

# --- TELEGRAM HANDLERS ---
@bot.message_handler(commands=['start'])
def send_welcome(message):
    bot.reply_to(message, WELCOME)

@bot.message_handler(func=lambda message: True)
def chat_with_gemini(message):
    try:
        response = get_gemini_response(message.text)
        bot.reply_to(message, response)
    except Exception as e:
        bot.reply_to(message, "Sorry, error aaya bhai. Try again later.")
